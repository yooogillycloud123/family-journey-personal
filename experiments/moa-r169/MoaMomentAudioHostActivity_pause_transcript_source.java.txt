 private File dayDir(String day){File d=new File(getFilesDir(),"moa-original-audio/"+day);if(!d.exists())d.mkdirs();return d;}
 private void record(String day){
  if(!trusted()||!validDay(day)){send("record_error","내부 일기 날짜가 유효하지 않습니다.");return;}
  if(recorder!=null){send("record_error","이미 녹음 중입니다.");return;}
  cancelSpeech();
  File f=new File(dayDir(day),System.currentTimeMillis()+"-"+UUID.randomUUID().toString().replace("-","")+".m4a");
  MediaRecorder r=null;
  try{
   r=new MediaRecorder();r.setAudioSource(MediaRecorder.AudioSource.MIC);
   r.setOutputFormat(MediaRecorder.OutputFormat.MPEG_4);r.setAudioEncoder(MediaRecorder.AudioEncoder.AAC);
   r.setAudioEncodingBitRate(96000);r.setAudioSamplingRate(44100);r.setOutputFile(f.getAbsolutePath());
   r.prepare();r.start();recorder=r;currentFile=f;recordStart=System.currentTimeMillis();recordDay=day;
   send("record_started","녹음 중입니다. 종료·저장을 눌러 주세요.");
  }catch(Exception e){
   try{if(r!=null)r.release();}catch(Exception ignored){}
   if(f.exists())f.delete();
   recorder=null;currentFile=null;send("record_error","Android 녹음 시작 실패: "+e.getClass().getSimpleName()+": "+e.getMessage());
  }
 }
 private void stopRecord(){
  MediaRecorder r=recorder;File f=currentFile;
  recorder=null;currentFile=null;

  boolean valid=false;try{r.stop();valid=f!=null&&f.exists()&&f.length()>0;}catch(RuntimeException e){send("record_error","녹음 종료 실패: "+e.getMessage());}

  send("record_saved","날짜별 음성파일 저장 완료 · "+((System.currentTimeMillis()-recordStart)/1000)+"초");
 }

    activePlayback="";
   MediaPlayer p=new MediaPlayer();p.setDataSource(f.getAbsolutePath());
   p.setOnCompletionListener(mp->{mp.release();if(player==mp){player=null;activePlayback="";}send("play_finished","재생 완료");});
   p.prepare();p.start();player=p;activePlayback=day+"/"+id;send("play_started","녹음 재생 중");

 }

  private void startSpeech(String token){
  if(!trusted()||token==null||token.length()>128)return;
  cancelSpeech();recognitionToken=token;
  if(!SpeechRecognizer.isRecognitionAvailable(this)){speechEvent(token,"기기 음성인식 서비스가 없습니다.","error");speechEvent(token,"","finished");return;}
  try{
   recognizer=SpeechRecognizer.createSpeechRecognizer(this);
   recognizer.setRecognitionListener(new RecognitionListener(){
    public void onReadyForSpeech(Bundle params){speechEvent(token,"","listening");}
    public void onBeginningOfSpeech(){}
    public void onRmsChanged(float rms){}
    public void onBufferReceived(byte[] buf){}
    public void onEndOfSpeech(){}
    public void onError(int error){speechEvent(token,"음성 인식 오류 "+error,"error");speechEvent(token,"","finished");cancelSpeech();}
    private void publish(Bundle b,String phase){
     ArrayList<String> a=b==null?null:b.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION);
     if(a!=null&&!a.isEmpty())speechEvent(token,a.get(0),phase);
    }
    public void onResults(Bundle results){publish(results,"success");speechEvent(token,"","finished");cancelSpeech();}
    public void onPartialResults(Bundle partial){publish(partial,"partial");}
    public void onEvent(int event,Bundle b){}
   });
   Intent i=new Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
   i.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL,RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
   i.putExtra(RecognizerIntent.EXTRA_LANGUAGE,"ko-KR");i.putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS,true);
   recognizer.startListening(i);
  }catch(Exception e){speechEvent(token,"음성 인식 시작 실패: "+e.getMessage(),"error");speechEvent(token,"","finished");cancelSpeech();}
 }

 private void startFileSpeech(String d,String id){
  if(!trusted()||!validDay(d)||!validId(id))return;
  if(fileSpeechBusy){send("transcribe_error","다른 파일의 글 변환이 진행 중입니다.");return;}
  if(recorder!=null){send("transcribe_error","녹음을 종료한 후 글로 변환해 주세요.");return;}
  final java.io.File source=new java.io.File(dayDir(d),id);
  if(!source.exists()||source.length()==0){send("transcribe_error","음성파일을 찾지 못했습니다.");return;}
  if(android.os.Build.VERSION.SDK_INT<33){send("transcribe_error","Android 13 이상에서 지원하는 기능입니다.");return;}
  if(!android.speech.SpeechRecognizer.isRecognitionAvailable(this)){send("transcribe_error","사용 가능한 기기 음성인식 서비스가 없습니다.");return;}
  cancelSpeech();fileSegments.clear();fileSpeechBusy=true;fileSpeechTarget=d+"/"+id;
  try{
   android.os.ParcelFileDescriptor[] pair=android.os.ParcelFileDescriptor.createPipe();
   fileRead=pair[0];fileWrite=pair[1];
   fileSpeech=android.speech.SpeechRecognizer.createSpeechRecognizer(this);
   final String target=fileSpeechTarget;
   fileSpeech.setRecognitionListener(new android.speech.RecognitionListener(){
    public void onReadyForSpeech(android.os.Bundle b){}
    public void onBeginningOfSpeech(){}
    public void onRmsChanged(float v){}
    public void onBufferReceived(byte[] b){}
    public void onEndOfSpeech(){}
    public void onError(int code){
     if(!target.equals(fileSpeechTarget))return;
     sendTranscript("transcribe_error",d,id,"저장된 음성 인식 실패("+code+"). 음성인식 서비스에서 파일 입력을 지원하지 않을 수 있습니다.");
     endFileSpeech();
    }
    public void onResults(android.os.Bundle b){
     if(!target.equals(fileSpeechTarget))return;
     java.util.ArrayList<String> list=b==null?null:b.getStringArrayList(android.speech.SpeechRecognizer.RESULTS_RECOGNITION);
     String recognized=(list!=null&&!list.isEmpty())?list.get(0):String.join(" ",fileSegments);
     if(recognized.trim().isEmpty())sendTranscript("transcribe_error",d,id,"인식된 문장이 없습니다.");
     else sendTranscript("transcribe_result",d,id,recognized.trim());
     endFileSpeech();
    }
    public void onPartialResults(android.os.Bundle b){}
    public void onEvent(int code,android.os.Bundle b){}
    public void onSegmentResults(android.os.Bundle b){
     java.util.ArrayList<String> x=b==null?null:b.getStringArrayList(android.speech.SpeechRecognizer.RESULTS_RECOGNITION);
     if(x!=null&&!x.isEmpty())fileSegments.add(x.get(0));
    }
    public void onEndOfSegmentedSession(){
     if(!target.equals(fileSpeechTarget))return;
     String combined=String.join(" ",fileSegments).trim();
     if(combined.isEmpty())sendTranscript("transcribe_error",d,id,"인식된 문장이 없습니다.");
     else sendTranscript("transcribe_result",d,id,combined);
     endFileSpeech();
    }
   });
   android.content.Intent intent=new android.content.Intent(android.speech.RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
   intent.putExtra(android.speech.RecognizerIntent.EXTRA_LANGUAGE_MODEL,android.speech.RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
   intent.putExtra(android.speech.RecognizerIntent.EXTRA_LANGUAGE,"ko-KR");
   intent.putExtra(android.speech.RecognizerIntent.EXTRA_AUDIO_SOURCE,fileRead);
   intent.putExtra(android.speech.RecognizerIntent.EXTRA_AUDIO_SOURCE_CHANNEL_COUNT,1);
   intent.putExtra(android.speech.RecognizerIntent.EXTRA_AUDIO_SOURCE_ENCODING,android.media.AudioFormat.ENCODING_PCM_16BIT);
   intent.putExtra(android.speech.RecognizerIntent.EXTRA_AUDIO_SOURCE_SAMPLING_RATE,16000);
   intent.putExtra(android.speech.RecognizerIntent.EXTRA_SEGMENTED_SESSION,android.speech.RecognizerIntent.EXTRA_AUDIO_SOURCE);
   fileSpeech.startListening(intent);
   sendTranscript("transcribe_started",d,id,"저장된 파일을 기기 음성인식 서비스에 전달하고 있습니다.");
   final android.os.ParcelFileDescriptor destination=fileWrite;
   new Thread(()->{
    try{streamPcm16k(source,destination,target);}
    catch(Exception e){runOnUiThread(()->{
      if(target.equals(fileSpeechTarget)){sendTranscript("transcribe_error",d,id,"오디오 디코딩 실패: "+e.getMessage());endFileSpeech();}
    });}
   },"moa-audio-file-transcriber").start();
  }catch(Exception e){
   sendTranscript("transcribe_error",d,id,"음성 변환 시작 실패: "+e.getClass().getSimpleName()+": "+e.getMessage());
   endFileSpeech();
  }
 }
 private void streamPcm16k(java.io.File source,android.os.ParcelFileDescriptor fd,String target)throws Exception{
  android.media.MediaExtractor extractor=new android.media.MediaExtractor();
  android.media.MediaCodec decoder=null;
  try(java.io.OutputStream output=new android.os.ParcelFileDescriptor.AutoCloseOutputStream(fd)){
   extractor.setDataSource(source.getAbsolutePath());
   int track=-1;android.media.MediaFormat format=null;
   for(int i=0;i<extractor.getTrackCount();i++){
    android.media.MediaFormat m=extractor.getTrackFormat(i);
    String type=m.getString(android.media.MediaFormat.KEY_MIME);
    if(type!=null&&type.startsWith("audio/")){track=i;format=m;break;}
   }
   if(track<0||format==null)throw new java.io.IOException("M4A 오디오 트랙이 없습니다.");
   extractor.selectTrack(track);
   decoder=android.media.MediaCodec.createDecoderByType(format.getString(android.media.MediaFormat.KEY_MIME));
   decoder.configure(format,null,null,0);decoder.start();
   android.media.MediaCodec.BufferInfo bufferInfo=new android.media.MediaCodec.BufferInfo();
   boolean inputDone=false,outputDone=false;
   long outSamples=0,started=android.os.SystemClock.elapsedRealtime();
   long accumulator=0;
   int sampleRate=format.containsKey(android.media.MediaFormat.KEY_SAMPLE_RATE)?format.getInteger(android.media.MediaFormat.KEY_SAMPLE_RATE):44100;
   int channels=format.containsKey(android.media.MediaFormat.KEY_CHANNEL_COUNT)?format.getInteger(android.media.MediaFormat.KEY_CHANNEL_COUNT):1;
   while(!outputDone&&fileSpeechBusy&&target.equals(fileSpeechTarget)){
    if(!inputDone){
     int index=decoder.dequeueInputBuffer(12000);
     if(index>=0){
      java.nio.ByteBuffer input=decoder.getInputBuffer(index);input.clear();
      int length=extractor.readSampleData(input,0);
      if(length<0){decoder.queueInputBuffer(index,0,0,0,android.media.MediaCodec.BUFFER_FLAG_END_OF_STREAM);inputDone=true;}
      else{decoder.queueInputBuffer(index,0,length,extractor.getSampleTime(),0);extractor.advance();}
     }
    }
    int outputIndex=decoder.dequeueOutputBuffer(bufferInfo,12000);
    if(outputIndex==android.media.MediaCodec.INFO_OUTPUT_FORMAT_CHANGED){
     android.media.MediaFormat fmt=decoder.getOutputFormat();
     sampleRate=fmt.containsKey(android.media.MediaFormat.KEY_SAMPLE_RATE)?fmt.getInteger(android.media.MediaFormat.KEY_SAMPLE_RATE):sampleRate;
     channels=fmt.containsKey(android.media.MediaFormat.KEY_CHANNEL_COUNT)?fmt.getInteger(android.media.MediaFormat.KEY_CHANNEL_COUNT):channels;
     int encoding=fmt.containsKey(android.media.MediaFormat.KEY_PCM_ENCODING)?fmt.getInteger(android.media.MediaFormat.KEY_PCM_ENCODING):android.media.AudioFormat.ENCODING_PCM_16BIT;
     if(encoding!=android.media.AudioFormat.ENCODING_PCM_16BIT)throw new java.io.IOException("지원되지 않는 PCM 인코딩");
    }else if(outputIndex>=0){
     java.nio.ByteBuffer decoded=decoder.getOutputBuffer(outputIndex);
     if(decoded!=null&&bufferInfo.size>0){
      decoded.position(bufferInfo.offset);decoded.limit(bufferInfo.offset+bufferInfo.size);
      decoded.order(java.nio.ByteOrder.LITTLE_ENDIAN);
      int frames=bufferInfo.size/Math.max(2,channels*2);
      java.io.ByteArrayOutputStream block=new java.io.ByteArrayOutputStream(frames*2);
      for(int i=0;i<frames;i++){
       int total=0;for(int ch=0;ch<channels;ch++)total+=decoded.getShort();
       short sample=(short)(total/channels);
       accumulator+=16000;
       if(accumulator>=sampleRate){
        accumulator-=sampleRate;block.write(sample&255);block.write((sample>>>8)&255);outSamples++;
       }
      }
      byte[] bytes=block.toByteArray();if(bytes.length>0){output.write(bytes);output.flush();}
      long expectedMs=outSamples*1000/16000;
      long wait=expectedMs-(android.os.SystemClock.elapsedRealtime()-started);
      if(wait>10)Thread.sleep(Math.min(wait,700));
     }
     outputDone=(bufferInfo.flags&android.media.MediaCodec.BUFFER_FLAG_END_OF_STREAM)!=0;
     decoder.releaseOutputBuffer(outputIndex,false);
    }
   }
  }finally{
   extractor.release();
   try{if(decoder!=null){decoder.stop();decoder.release();}}catch(Exception ignored){}
  }
 }

  private android.net.Uri safeVideoUri(String videoId){
   if(!trusted()||videoId==null||!videoId.matches("v[0-9]{1,18}"))return null;
   try{
     long mediaId=Long.parseLong(videoId.substring(1));
     android.net.Uri uri=android.content.ContentUris.withAppendedId(android.provider.MediaStore.Video.Media.EXTERNAL_CONTENT_URI,mediaId);
     try(android.database.Cursor c=getContentResolver().query(uri,new String[]{android.provider.MediaStore.Video.Media._ID},null,null,null)){
       if(c!=null&&c.moveToFirst())return uri;
     }
   }catch(Exception ignored){}
   return null;
  }
  /** Optional HTML5 video source. Browser may reject a content:// URL, so JS must fallback safely. */

   android.net.Uri uri=safeVideoUri(videoId);return uri==null?"":uri.toString();
  }
  /** Explicit user-tap capture from this source video; stores a NEW JPEG without editing original. */
  @JavascriptInterface public void captureVideoFrame(String videoId,long millis){
   android.net.Uri src=safeVideoUri(videoId);
   if(src==null){main.post(()->send("video_capture_error","원본 동영상을 찾을 수 없습니다."));return;}
   new Thread(()->{
     android.media.MediaMetadataRetriever retriever=new android.media.MediaMetadataRetriever();
     android.net.Uri dest=null;boolean published=false;
     try{
       retriever.setDataSource(MoaMomentAudioHostActivity.this,src);
       android.graphics.Bitmap bitmap=retriever.getFrameAtTime(Math.max(0,millis)*1000,android.media.MediaMetadataRetriever.OPTION_CLOSEST_SYNC);
       if(bitmap==null)throw new java.io.IOException("해당 시각의 화면을 추출하지 못했습니다.");
       android.content.ContentValues values=new android.content.ContentValues();
       values.put(android.provider.MediaStore.Images.Media.DISPLAY_NAME,"MOA_video_"+System.currentTimeMillis()+".jpg");
       values.put(android.provider.MediaStore.Images.Media.MIME_TYPE,"image/jpeg");
       if(android.os.Build.VERSION.SDK_INT>=29){
         values.put(android.provider.MediaStore.Images.Media.RELATIVE_PATH,android.os.Environment.DIRECTORY_PICTURES+"/MOA-Video-Captures/");
         values.put(android.provider.MediaStore.Images.Media.IS_PENDING,1);
       }
       android.content.ContentResolver cr=getContentResolver();
       dest=cr.insert(android.provider.MediaStore.Images.Media.EXTERNAL_CONTENT_URI,values);
       if(dest==null)throw new java.io.IOException("사진 저장 공간을 열지 못했습니다.");
       try(java.io.OutputStream out=cr.openOutputStream(dest,"w")){
         if(out==null||!bitmap.compress(android.graphics.Bitmap.CompressFormat.JPEG,94,out))
           throw new java.io.IOException("JPEG 저장 실패");
       }finally{bitmap.recycle();}
       if(android.os.Build.VERSION.SDK_INT>=29){
         android.content.ContentValues ready=new android.content.ContentValues();
         ready.put(android.provider.MediaStore.Images.Media.IS_PENDING,0);
         if(cr.update(dest,ready,null,null)<1)throw new java.io.IOException("캡처 공개 처리 실패");
       }
       published=true;
       main.post(()->send("video_capture_saved","Pictures/MOA-Video-Captures에 현재 화면 저장 완료"));
     }catch(Exception e){
       final String err=e.getMessage();main.post(()->send("video_capture_error","동영상 캡처 실패: "+err));
     }finally{
       try{retriever.release();}catch(Exception ignored){}
       if(!published&&dest!=null)try{getContentResolver().delete(dest,null,null);}catch(Exception ignored){}
     }
   },"moa-r169-video-frame").start();
  }

  @JavascriptInterface public String captureTranscripts(String day){
   if(!trusted()||!validDay(day))return "{}";
   org.json.JSONObject found=new org.json.JSONObject();
   try{
    java.util.Map<String,?> entries=getSharedPreferences("moa_camera_native_v1",MODE_PRIVATE).getAll();
    for(java.util.Map.Entry<String,?> ent:entries.entrySet()){
      if(!ent.getKey().startsWith("m:")||!(ent.getValue() instanceof String))continue;
      org.json.JSONObject record;
      try{record=new org.json.JSONObject((String)ent.getValue());}catch(Exception ex){continue;}
      String id=record.optString("voiceFileId","");
      if(day.equals(record.optString("voiceDay",""))&&validId(id)){
        org.json.JSONObject m=new org.json.JSONObject();
        m.put("autoTranscript",record.optString("voiceAutoTranscript",""));
        m.put("autoStatus",record.optString("voiceAutoStatus",""));
        m.put("captureNote",record.optString("note",""));
        found.put(id,m);
      }
      org.json.JSONArray history=record.optJSONArray("voiceRecords");
      if(history==null)continue;
      for(int n=0;n<history.length();n++){
        org.json.JSONObject item=history.optJSONObject(n);if(item==null)continue;
        String file=item.optString("id","");
        if(!validId(file)||!day.equals(item.optString("day","")))continue;
        org.json.JSONObject data=found.optJSONObject(file);
        if(data==null)data=new org.json.JSONObject();
        String transcript=item.optString("autoTranscript","");
        if(!transcript.isEmpty()||!data.has("autoTranscript"))data.put("autoTranscript",transcript);
        data.put("autoStatus",item.optString("autoStatus",""));
        if(!data.has("captureNote"))data.put("captureNote",record.optString("note",""));
        found.put(file,data);
      }
    }
   }catch(Exception ignored){}
   return found.toString();
  }
